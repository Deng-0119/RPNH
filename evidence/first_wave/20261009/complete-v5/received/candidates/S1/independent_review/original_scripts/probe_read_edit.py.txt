"""Real Registry comparison of existing variable read and new static read.
Only adapt the test lowerer in memory; no product source or stored PN is edited.
"""
import inspect,sys,runpy,tempfile
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]/'rpnh-static-lease-read-fix/source/tests'))
import test_static_lease_interactions as fixture
from cpn.rpnh.registry.module_runtime import hydrate_module_runtime
from cpn.rpnh.registry.schema_catalog import canonical_json
source=inspect.getsource(fixture._world)
source=source.replace('same_pool=True,','same_pool=True, reader_mode="static",')
needle='        return replace(fragment,'
insert='''        if with_edit:
            places.append(PlaceDeclaration("reader_control", TEXT, channel="control"))
            transitions=(*transitions,TransitionDeclaration("read", "read"))
            operations=(*operations,replace(operations[0],name="read",inputs=(),outputs=(),request_port=None,outcomes=(OutcomeDeclaration("complete",()),)))
            arcs=(*arcs,ArcDeclaration("reader_control","run","output",mode="produce",outcome="complete",emit="content_less"),ArcDeclaration("reader_control","read","input"))
            if reader_mode=="static":
                arcs=(*arcs,ArcDeclaration("variable","read","input",mode="read"))
            else:
                variables=(*variables,VariableResourceArc("read","reader_control","variable",(LeaseClaimTemplate("document",access_mode="read"),)))
'''
source=source.replace(needle,insert+needle)
source=source.replace('document = _simple_module("LeaseInteractions").to_dict()','document = _simple_module("LeaseInteractions").to_dict()\n    document["budget_buckets"][0]["max_attempts"]=5')
source=source.replace('(TEXT,), 3, 0, 3, 0)','(TEXT,), 5, 0, 5, 0)')
ns=dict(vars(fixture));exec(compile(source,'<review-test-lowerer>','exec'),ns)
world=ns['_world']

def slot(o):
 return next(t for t in hydrate_module_runtime(o._core)[2].tokens if t.state.place=='step.variable')

for mode in ('variable','static'):
 for order in ('read-first','edit-first'):
  with tempfile.TemporaryDirectory(prefix='static-lease-read-edit-') as tmp:
   o=world(Path(tmp)/'run',variable_mode='produce',with_edit=True,static_read=False,reader_mode=mode)
   produce=o.admit('step.run',logical_tau=1,command_id='review:produce:admit')
   fixture._settle(o,produce)
   initial=slot(o)
   first='read' if order=='read-first' else 'edit'
   second='edit' if order=='read-first' else 'read'
   a=o.admit('step.'+first,logical_tau=2,command_id='review:'+first+':admit');assert a is not None
   e=o.start(a,command_id='review:'+first+':start')
   assert o.admit('step.'+second,logical_tau=3,command_id='review:'+second+':blocked') is None
   p=o.products(e,outcome_id='complete',products={},command_id='review:'+first+':products');o.succeed(p,command_id='review:'+first+':success')
   intermediate=slot(o)
   if first=='read':
    assert (intermediate.token_ref,intermediate.state)==(initial.token_ref,initial.state)
   else:
    assert intermediate.token_ref!=initial.token_ref
    assert intermediate.state.resource_ref==initial.state.resource_ref
   a=o.admit('step.'+second,logical_tau=3,command_id='review:'+second+':admit');assert a is not None
   e=o.start(a,command_id='review:'+second+':start')
   p=o.products(e,outcome_id='complete',products={},command_id='review:'+second+':products');o.succeed(p,command_id='review:'+second+':success')
   current=slot(o)
   assert current.token_ref!=initial.token_ref
   assert current.state.lease_identity_ref==initial.state.lease_identity_ref
   assert current.state.resource_ref==initial.state.resource_ref
   if second=='read':
    assert (current.token_ref,current.state)==(intermediate.token_ref,intermediate.state)
   print(mode,order,'PASS: concurrent opposite access rejected; later access admitted; read retains exact token; edit returns new token with same identity/RVR',flush=True)
