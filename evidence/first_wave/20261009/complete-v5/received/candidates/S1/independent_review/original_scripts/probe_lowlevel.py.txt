"""Independent direct Registry probes; run with target source on PYTHONPATH."""
from pathlib import Path
from tempfile import TemporaryDirectory
import runpy,sys
from dataclasses import replace
from cpn.rpnh.registry.module_runtime import hydrate_module_runtime
from cpn.rpnh.registry.invocations import FiringClaim, InvocationLifecycle
from cpn.rpnh.registry.strict_contracts import _registered
from cpn.rpnh.registry.publication import _version_from_payload

scope=runpy.run_path(str(Path(__file__).resolve().parents[1]/'rpnh-static-lease-read-fix/source/tests/test_static_lease_reads.py'))

def make_claim(owner, mode):
    core=owner._core
    executable,structure,marking=hydrate_module_runtime(core)
    transition=next(t for t in executable.transitions if t.transition_id=='step.run')
    _,root=_registered(core,executable.team_design_root_ref,'team_design_root/v1')
    task_round=_version_from_payload(root['task_round_ref'])
    _,round_data=_registered(core,task_round,'task_round/v1')
    all_refs=tuple(sorted((t.token_ref for t in marking.tokens),key=lambda r:(r.entity_type,str(r.entity_id),str(r.version_id))))
    leases={t.token_ref for t in marking.tokens if t.state.place=='step.lease'}
    consumed_places={p for p,t,_ in structure.token_input_arcs if t=='step.run'}
    consumed=tuple(t.token_ref for t in marking.tokens if t.state.place in consumed_places)
    if mode=='missing_lease': refs=tuple(r for r in all_refs if r not in leases)
    else: refs=all_refs
    if mode=='consume_lease': consumed=tuple(set(consumed)|leases)
    return FiringClaim(task_ref=_version_from_payload(root['task_ref']),task_branch_ref=_version_from_payload(round_data['task_branch_ref']),task_round_ref=task_round,net_instance_ref=executable.net_ref,plan_ref=_version_from_payload(_registered(core,executable.net_ref,'net_instance/v1')[1]['plan_ref']),node_ref=transition.node_ref,operation_binding_ref=transition.operation_binding_ref,marking_checkpoint_ref=marking.checkpoint_ref,principal_ref=transition.principal_ref,logical_tau=0,attempt_index=1,claimed_input_refs=refs,consumed_input_refs=consumed,activation_ref=transition.activation_ref,agent_ref=transition.agent_ref)

if __name__=='__main__':
    import cpn
    print('source',cpn.__file__,flush=True)
    for mode in ['correct','missing_lease','consume_lease']:
        with TemporaryDirectory(prefix='static-lease-lowlevel-') as tmp:
            owner=scope['lease_owner'](Path(tmp)/'run')
            try:
                claim=make_claim(owner,mode)
                result=InvocationLifecycle(owner._core).admit_firing(claim,idempotency_key=f'review:{mode}')
                print(mode,'ADMITTED',result.context.own_transition_firing_ref,flush=True)
            except Exception as exc: print(mode,'REJECTED',type(exc).__name__,str(exc),flush=True)
